    // Load persisted state from localStorage with safe fallback
    function loadState() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (!raw) return JSON.parse(JSON.stringify(DEFAULT_STATE));
        const parsed = JSON.parse(raw);
        const mergedProducts =
          Array.isArray(parsed.products) && parsed.products.length
            ? parsed.products.map((prod) => {
                const def = DEFAULT_STATE.products.find((d) => d.id === prod.id);
                const hasCustomUrl =
                  prod.logoUrl &&
                  typeof prod.logoUrl === 'string' &&
                  !prod.logoUrl.includes('placehold.co');
                const hasCustomBanner =
                  prod.imageUrl &&
                  typeof prod.imageUrl === 'string' &&
                  !prod.imageUrl.includes('placehold.co');
                return {
                  ...prod,
                  logoUrl: hasCustomUrl ? prod.logoUrl : (def ? def.logoUrl : ITEM_LOGOS.nitroBasic),
                  imageUrl: hasCustomBanner ? prod.imageUrl : (def ? def.imageUrl : ITEM_BANNERS.nitroBasic),
                };
              })
            : DEFAULT_STATE.products;
        return {
          ...JSON.parse(JSON.stringify(DEFAULT_STATE)),
          ...parsed,
          robux: {
            ...JSON.parse(JSON.stringify(DEFAULT_STATE.robux)),
            ...(parsed.robux || {}),
          },
          products: mergedProducts,
          paymentMethods:
            Array.isArray(parsed.paymentMethods) && parsed.paymentMethods.length
              ? parsed.paymentMethods
              : DEFAULT_STATE.paymentMethods,
          orders: Array.isArray(parsed.orders) ? parsed.orders : DEFAULT_STATE.orders,
          users: Array.isArray(parsed.users) && parsed.users.length ? parsed.users : DEFAULT_STATE.users,
          cart: Array.isArray(parsed.cart) ? parsed.cart : [],
        };
      } catch (err) {
        console.error('Failed to load state:', err);
        return JSON.parse(JSON.stringify(DEFAULT_STATE));
      }
    }

    let state = loadState();

    // Format numeric amounts into Bangladeshi Taka (BDT)
    function formatBDT(amount) {
      const num = Number(amount || 0);
      const formatted = Number.isInteger(num) ? num.toLocaleString('en-US') : num.toFixed(2);
      return `৳${formatted}`;
    }

    // Compute current Robux availability status
    function getRobuxStockStatus() {
      const stock = Number(state.robux.robuxStock || 0);
      const lowThreshold = Number(state.robux.lowStockThreshold || 0);
      if (state.robux.forceOutOfStock || stock <= 0) {
        return { available: false, label: 'Out of stock', tone: 'rose' };
      }
      if (stock <= lowThreshold) {
        return { available: true, label: 'Low stock', tone: 'amber' };
      }
      return { available: true, label: 'In stock', tone: 'emerald' };
    }

    // Format countdown string until configured restock timestamp
    function getRestockCountdownText() {
      if (!state.robux.restockAt) return '';
      const target = new Date(state.robux.restockAt).getTime();
      if (!Number.isFinite(target)) return '';
      const diffMs = target - (state.nowTick || Date.now());
      if (diffMs <= 0) return 'Restocking Soon';
      const totalSec = Math.floor(diffMs / 1000);
      const hrs = String(Math.floor(totalSec / 3600)).padStart(2, '0');
      const mins = String(Math.floor((totalSec % 3600) / 60)).padStart(2, '0');
      const secs = String(totalSec % 60).padStart(2, '0');
      return `${hrs}h : ${mins}m : ${secs}s`;
    }

    function getCartCount() {
      return state.cart.reduce((sum, item) => sum + (item.itemType === 'robux' ? 1 : Number(item.quantity || 1)), 0);
    }

    function getCartTotalBDT() {
      return state.cart.reduce((sum, item) => sum + Math.round(Number(item.unitPriceBdt) * Number(item.quantity)), 0);
    }

    function addToCart(newItem) {
      const existing = state.cart.find((i) => i.key === newItem.key);
      if (existing && newItem.itemType !== 'robux') {
        existing.quantity = Number(existing.quantity) + Number(newItem.quantity || 1);
      } else {
        state.cart.push(newItem);
      }
      saveState();
      showToast(`${newItem.title} added to cart!`, 'success');
      render();
    }

    function removeFromCart(key) {
      state.cart = state.cart.filter((i) => i.key !== key);
      saveState();
      render();
    }

    // Role-based access control helpers
    function canAccessAdminTab(tab) {
      const role = state.activeRole;
      if (role === 'owner' || role === 'admin') return true;
      if (role === 'stock_manager') return tab === 'stock' || tab === 'marketplace';
      if (role === 'order_manager') return tab === 'orders';
      return false;
    }

    }

    function canAssignRoles() {
      return state.activeRole === 'owner' || state.activeRole === 'admin';

    function isCurrentUserAuthorizedOwner() {
      const authorizedEmail = (state.exclusiveOwnerEmail || 'owner@bluevolt.bd').trim().toLowerCase();
      if (!state.currentUser || !state.currentUser.email) {
        return state.activeRole === 'owner';
      }
      return state.currentUser.email.trim().toLowerCase() === authorizedEmail;
    }

    function canReceiveOwnerRole(targetEmail) {
      if (!state.exclusiveOwnerOnlyMe) return true;
      const authorizedEmail = (state.exclusiveOwnerEmail || 'owner@bluevolt.bd').trim().toLowerCase();
      return String(targetEmail || '').trim().toLowerCase() === authorizedEmail;
    }

    function getAvailableRolesToAssign() {
      if (state.activeRole === 'owner') {
        const canGiveOwner = !state.exclusiveOwnerOnlyMe || isCurrentUserAuthorizedOwner();
        return canGiveOwner
          ? ['owner', 'admin', 'stock_manager', 'order_manager']
          : ['admin', 'stock_manager', 'order_manager'];
      }
      if (state.activeRole === 'admin') {
        return ['admin', 'stock_manager', 'order_manager'];
      }
      return [];
    }

    // Navigation helper
    function navigate(route, params = {}) {
      state.activeRoute = route;
      state.quickMenuOpen = false;
      if (params.category) state.selectedCategory = params.category;
      if (params.productId) state.selectedProductId = params.productId;
      if (route === 'admin') {
        const staffRoles = ['owner', 'admin', 'stock_manager', 'order_manager'];
        if (!staffRoles.includes(state.activeRole)) {
          state.activeRole = 'owner';
        }
        if (!canAccessAdminTab(state.activeAdminTab)) {

    function escapeHtml(str) {
      return String(str ?? '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');

    function submitManualOrder(e) {
      e.preventDefault();
      if (state.cart.length === 0) {
        showToast('Your cart is empty.', 'error');
        return;
      }
      const enabledMethods = state.paymentMethods.filter((m) => m.isEnabled);
      const selectedMethodId = window.__selectedPaymentMethodId || (enabledMethods[0] && enabledMethods[0].id);
      const method = enabledMethods.find((m) => m.id === selectedMethodId);
      if (!method) {
        showToast('Please select an enabled payment method.', 'error');
        return;
      }

      const trxId = (document.getElementById('checkoutTrxId')?.value || '').trim().toUpperCase();
      const senderLast4 = (document.getElementById('checkoutLast4')?.value || '').trim();
      const robloxUser = (document.getElementById('checkoutRobloxUser')?.value || '').trim();
      const discordUser = (document.getElementById('checkoutDiscordUser')?.value || '').trim();

      if (trxId.length < 4) {
        showToast('Enter a valid Transaction ID (TrxID).', 'error');
        return;
      }
      if (!/^[0-9]{4}$/.test(senderLast4)) {
        showToast('Sender last 4 digits must be exactly 4 numeric digits.', 'error');
        return;
      }
      const duplicateTrx = state.orders.some((o) => o.trxId.toUpperCase() === trxId && o.status !== 'Rejected');
      if (duplicateTrx) {
        showToast('Duplicate Transaction ID detected. Each order requires a unique TrxID.', 'error');
        return;
      }

      const customerEmail = state.currentUser ? state.currentUser.email : 'guest@bluevolt.bd';
      const nowStr = new Date().toISOString().slice(0, 16).replace('T', ' ');

      for (const item of state.cart) {
        const lineTotal = Math.round(Number(item.unitPriceBdt) * Number(item.quantity));
        const orderNumber = 'BV-' + Math.random().toString(36).substring(2, 10).toUpperCase();
        state.orders.unshift({
          id: 'ord-' + Date.now() + '-' + Math.random().toString(36).slice(2, 5),
          orderNumber,
          customerEmail,
          itemType: item.itemType,
          productId: item.productId || null,
          itemTitle: item.title,
          quantity: Number(item.quantity),
          unitRateBdt: Number(item.unitPriceBdt),
          totalAmountBdt: lineTotal,
          paymentMethodId: method.id,
          paymentMethodName: method.name,
          trxId,
          senderLast4,
          robloxUsername: item.requiresRobloxUsername ? robloxUser : null,
          discordUsername: (item.requiresDiscordTag || item.category === 'Discord Items') ? discordUser : null,
          status: 'Pending',
          internalNote: '',
          createdAt: nowStr,
        });

        if (item.itemType === 'robux') {
          state.robux.robuxStock = Math.max(0, Number(state.robux.robuxStock) - Number(item.quantity));
        } else if (item.productId) {
          const prod = state.products.find((p) => p.id === item.productId);
          if (prod) {
            prod.stockQuantity = Math.max(0, Number(prod.stockQuantity) - Number(item.quantity));
          }
        }
      }

      state.cart = [];
      saveState();
      showToast('Order submitted! Status is now Pending verification.', 'success');
      navigate('orders');
    }

    async function handleAuthSubmit(e, mode) {
      e.preventDefault();
      const email = document.getElementById('authEmail').value.trim().toLowerCase();
      const password = document.getElementById('authPassword').value;
      const fullName = mode === 'signup' ? document.getElementById('authFullName').value.trim() : email.split('@')[0];

      if (supabaseClient) {
        try {
          const res =
            mode === 'signup'
              ? await supabaseClient.auth.signUp({ email, password, options: { data: { full_name: fullName } } })
              : await supabaseClient.auth.signInWithPassword({ email, password });
          if (res.error) {
            showToast(res.error.message, 'error');
            return;
          }
        } catch (err) {
          showToast('Supabase authentication request failed.', 'error');
          return;
        }
      }

      let existing = state.users.find((u) => u.email.toLowerCase() === email);
      if (!existing) {
        existing = { id: 'u-' + Date.now(), email, fullName, roles: [] };
        state.users.push(existing);
      }
      if (state.exclusiveOwnerOnlyMe && existing.roles.includes('owner') && !canReceiveOwnerRole(existing.email)) {
        existing.roles = existing.roles.filter((r) => r !== 'owner');
      }
      state.currentUser = { email: existing.email, fullName: existing.fullName };
      if (existing.roles.includes('owner') && canReceiveOwnerRole(existing.email)) state.activeRole = 'owner';
      else if (existing.roles.includes('admin')) state.activeRole = 'admin';
      else if (existing.roles.includes('stock_manager')) state.activeRole = 'stock_manager';
      else if (existing.roles.includes('order_manager')) state.activeRole = 'order_manager';
      else state.activeRole = 'customer';

      state.authModal = null;
      saveState();
      showToast(`Signed in as ${existing.email} (${state.activeRole}).`, 'success');
      render();
    }

    function copyText(text, successMsg) {
      try {
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(text).catch(() => {});
        }
        const ta = document.createElement('textarea');
        ta.value = text;
        ta.setAttribute('readonly', '');
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
      } catch {
        // Fallback if clipboard API is restricted
      }
      showToast(successMsg, 'success');
    }
